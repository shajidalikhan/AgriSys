\section{Introduction} % <-- Label for TOC navigation

\begin{frame}{Introduction: Next-Generation Agricultural Expert System}
	\footnotesize
	\begin{itemize}
		\item \textbf{The Shift:} Moving from static, rule-based apps to AI-driven conversational agents.
		
		\vspace{0.2cm}
		\item \textbf{The Challenge:} Farming requires synthesizing heterogeneous data (visual anomalies, complex soil chemistry, local policies).
		
		\vspace{0.2cm}
		\item \textbf{Current Limitations:} Generic LLMs hallucinate agronomic advice, lack domain specificity, and fail on low-quality field media.
		
		\vspace{0.2cm}
		\item \textbf{Proposed Solution:} A multimodal framework combining computer vision, dynamic knowledge retrieval, and strict domain constraints.
	\end{itemize}
\end{frame}


\section{Related Work}
\begin{frame}{Related Work -- Vision and Multimodality}
	\footnotesize
	\begin{itemize}
		\item \textbf{Multimodal LLMs (e.g., AgriM-LLM):} Attempt to fuse visual encoders with language models for disease management.
		\begin{itemize}
			\item \textit{Limitation:} Accuracy degrades heavily on the blurry, compressed media typical of real-world farming.
		\end{itemize}
		
		\vspace{0.2cm}
		\item \textbf{Diagnostic CNNs:} Proven highly accurate in identifying nutrient deficiencies via static plant imagery.
		\begin{itemize}
			\item \textit{Limitation:} Over-reliant on high-fidelity, static images rather than dynamic, noisy field video.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Related Work -- Architecture and NLP}
	\footnotesize
	\begin{itemize}
		\item \textbf{Retrieval-Augmented Generation (RAG):} The current standard for grounding LLMs in localized data to prevent hallucinations.
		\begin{itemize}
			\item \textit{Limitation:} Current RAG implementations lack temporal memory; they treat queries in isolation instead of tracking a full crop cycle.
		\end{itemize}
		
		\vspace{0.2cm}
		\item \textbf{Lexical Processing:} Standard NLP pipelines often fail to capture localized agricultural vernacular and context. 
		
		\vspace{0.2cm}
		\item \textbf{The Regulatory Void:} Existing chatbots offer chemical/agronomic advice while operating in a legal vacuum, completely bypassing agricultural laws and compliance frameworks.
	\end{itemize}
\end{frame}


\section{Research Gaps}
\begin{frame}{Identified Research Gaps}

	\footnotesize
	\begin{enumerate}
		\item \textbf{Low-Fidelity Media Handling:} Current models cannot accurately classify anomalies from the low-resolution, noisy images or videos captured by standard smartphones in the field.
		
		\vspace{0.2cm}
		\item \textbf{Contextual Memory Loss:} Existing RAG architectures fail to maintain a continuous, evolving state of a farm over an entire harvesting season.
		
		\vspace{0.2cm}
		\item \textbf{The Compliance Disconnect:} There is an absence of expert systems that cross-reference deep biochemical recommendations with regional agricultural regulations and legal constraints.
	\end{enumerate}
\end{frame}

\section{Objectives}
\begin{frame}{Research Objectives}

	\footnotesize
	\begin{enumerate}
		\item \textbf{Enhance Visual Input:} Integrate Super-Resolution (Image and Video) preprocessing to restore low-fidelity field media prior to diagnostic classification.
		
		\medskip
		\item \textbf{Develop Robust Architecture:} Construct an advanced RAG framework equipped with state-tracking to maintain full-season agronomic context.
		
		\medskip
		\item \textbf{Embed Multidisciplinary Intelligence:} Bridge agronomic advice with legal compliance, ensuring all soil and chemical recommendations adhere strictly to local policies.
	\end{enumerate}
\end{frame}
