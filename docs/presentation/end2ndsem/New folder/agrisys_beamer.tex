% ── SLIDE 2: INTRODUCTION ─────────────────────────────────
\begin{frame}{The Agricultural Advisory Crisis in India}
  \framesubtitle{Why existing systems fail the 146 million farmers who need them most}

  \vspace{-0.4em}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{itemize}[itemsep=1.2pt, topsep=0.8pt]\footnotesize
        \item India has \textbf{1 agricultural extension worker per 1{,}000+ farmers} — advice is inaccessible at scale \textcolor{Muted}{(ICAR, 2023)}
        \item Crop disease is the \textbf{leading cause of yield loss}; most farmers diagnose by eye, often too late \textcolor{Muted}{(FAO, 2022)}
        \item Soil nutrient deficiencies cost \textbf{20--30\% of potential yield}; soil testing adoption is $<$15\% \textcolor{Muted}{(NABARD, 2023)}
        \item Mandi price information asymmetry causes farmers to sell \textbf{30--40\% below optimal price} \textcolor{Muted}{(eNAM, 2023)}
        \item Existing AI tools are \textbf{English-only, text-only}, or require stable internet — excluding most target users
      \end{itemize}
    \end{column}

    \begin{column}{0.44\textwidth}
      \begin{tcolorbox}[agricard=Purple, title=What AI can offer, before skip=0pt, top=3pt, bottom=3pt]

        \begin{itemize}[leftmargin=1em, itemsep=0.5pt, topsep=0.5pt]

          \item[$\rightarrow$] Instant leaf-image disease diagnosis + XAI
          \item[$\rightarrow$] Soil-sensor-grounded nutrition recommendations
          \item[$\rightarrow$] Live mandi price advisory with sell/hold guidance
        \end{itemize}
      \end{tcolorbox}
      \vspace{2pt}
      \begin{tcolorbox}[agricard=Teal, title=Why now?, before skip=0pt, top=3pt, bottom=3pt]

% ── SLIDE 5: PROBLEM STATEMENT ────────────────────────────
\begin{frame}{Formal Problem Statement}
  \begin{tcolorbox}[psbox, top=4pt, bottom=4pt]
    \footnotesize Indian smallholder farmers face \textbf{simultaneous, interdependent challenges} --- crop disease, nutrient deficiency, uncertain crop selection, irrigation scheduling, and price realisation --- that no single AI model or modality can address. Existing expert systems are \textbf{text-only, single-agent, single-language, and non-explainable}, making them inaccessible and untrustworthy for the target population. \textbf{No current system} accepts voice in Hindi, diagnoses disease from leaf images, integrates soil sensor data, and advises on live mandi prices within a single conversational turn --- with explanations grounded in verified agronomic knowledge sources.
  \end{tcolorbox}

  \vspace{0.2em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{tcolorbox}[agricard=Coral, title=Challenges to overcome, before skip=0pt, top=3pt, bottom=3pt]
        \scriptsize
        \begin{itemize}[leftmargin=1.1em, itemsep=0.5pt, topsep=0pt]
          \item \textbf{Modality heterogeneity} — voice, image, text, sensor, price are structurally different inputs
          \item \textbf{Domain depth} — agriculture requires expert-level, crop- and region-specific knowledge
          \item \textbf{Hallucination risk} — LLMs fabricate crop-care advice with dangerous confidence
          \item \textbf{Language barrier} — Indic ASR accuracy sub-optimal for agricultural vocabulary
          \item \textbf{Trust} — farmers reject advice they cannot understand or verify
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{tcolorbox}[agricard=Green, title=Solution constraints, before skip=0pt, top=3pt, bottom=3pt]
        \scriptsize
        \begin{itemize}[leftmargin=1.1em, itemsep=0.5pt, topsep=0pt]
          \item Must work on commodity Android device + 3G connection
          \item Response time $<$4 s (cloud) $\cdot$ $<$8 s (PWA offline)
          \item Knowledge grounded exclusively in ICAR, FAO, CGIAR verified sources
          \item Explainability must be farmer-readable — visual, not numerical
          \item Multi-crop: wheat, rice, maize, cotton, tomato, potato and more
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}
